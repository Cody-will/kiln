
// ---- instruction builder ----

impl Cg {
	fun begin_fn(&mut self) {
		self.fb = string::new()
		self.allocas = string::new()
		self.nv = 0
		self.nb = 0
		self.open = false
		b := self.new_block()
		self.switch_to(b)
	}

	// The finished function text.
	fun end_fn(&mut self, header string) {
		if self.open {
			self.emit("unreachable".to_string())
			self.open = false
		}
		self.funcs.push(&header)
		self.funcs.push(" {\nentry:\n")
		self.funcs.push(&self.allocas)
		self.funcs.push("  br label %b0\n")
		self.funcs.push(&self.fb)
		self.funcs.push("}\n\n")
	}

	fun new_block(&mut self) u32 {
		n := self.nb
		self.nb += 1
		return n
	}

	fun switch_to(&mut self, b u32) {
		if self.open {
			self.fb.push("  unreachable\n")
		}
		self.fb.push("b{b}:\n")
		self.open = true
	}

	fun ensure_open(&mut self) {
		if !self.open {
			b := self.new_block()
			self.fb.push("b{b}:\n")
			self.open = true
		}
	}

	fun emit(&mut self, text string) {
		self.ensure_open()
		self.fb.push("  ")
		self.fb.push(&text)
		self.fb.push("\n")
	}

	fun term(&mut self, text string) {
		self.emit(text)
		self.open = false
	}

	fun tmp(&mut self) string {
		self.nv += 1
		return "%v{self.nv}"
	}

	fun inst(&mut self, t &str, text string) V {
		n := self.tmp()
		self.emit("{n} = {text}")
		return val(n, t)
	}

	fun iconst_t(&mut self, t &str, v i128) V {
		if is_float_t(t) {
			return val(f64_lit(v as f64), t)
		}
		// Normalize to the type's width (two's complement).
		b := bits(t)
		x mut := v
		if b < 128 {
			m := 1i128 << (b as i128)
			x = x % m
			if x < 0 {
				x += m
			}
			if x >= m / 2 {
				x -= m
			}
		}
		return val("{x}", t)
	}

	fun iconst(&mut self, v i64) V {
		return val("{v}", "i64")
	}

	fun f64const(&mut self, x f64) V {
		return val(f64_lit(x), "double")
	}

	fun f32const(&mut self, x f32) V {
		return val(f64_lit(x as f64), "float")
	}

	fun ptr_of(&mut self, addr &V, off i64) string {
		a := if off != 0 { self.iadd_imm(addr, off) } else { addr.clone() }
		p := self.tmp()
		self.emit("{p} = inttoptr i64 {a.n} to ptr")
		return p
	}

	fun load(&mut self, t &str, addr &V, off i64) V {
		p := self.ptr_of(addr, off)
		return self.inst(t, "load {t}, ptr {p}, align 1")
	}

	fun store(&mut self, v &V, addr &V, off i64) {
		p := self.ptr_of(addr, off)
		self.emit("store {v.t} {v.n}, ptr {p}, align 1")
	}

	fun copy(&mut self, dst &V, src &V, size u32) {
		if size == 0 {
			return
		}
		if size == 1 || size == 2 || size == 4 || size == 8 {
			t := int_of_bits(size * 8)
			v := self.load(&t, src, 0)
			self.store(&v, dst, 0)
			return
		}
		self.declare("llvm.memmove.p0.p0.i64", "declare void @llvm.memmove.p0.p0.i64(ptr, ptr, i64, i1)")
		d := self.ptr_of(dst, 0)
		s := self.ptr_of(src, 0)
		self.emit("call void @llvm.memmove.p0.p0.i64(ptr {d}, ptr {s}, i64 {size}, i1 false)")
	}

	fun slot(&mut self, size u32, align u32) V {
		self.nv += 1
		s := "%s{self.nv}"
		n := if size == 0 { 1 } else { size }
		a := if align == 0 { 1 } else { align }
		self.allocas.push("  {s} = alloca [{n} x i8], align {a}\n")
		self.nv += 1
		v := "%v{self.nv}"
		self.allocas.push("  {v} = ptrtoint ptr {s} to i64\n")
		return val(v, "i64")
	}

	fun slot_for(&mut self, t &Ty) V {
		l := self.layout(t)
		return self.slot(l.size, l.align)
	}

	// ---- arithmetic ----

	fun bin(&mut self, op &str, a &V, b &V) V {
		return self.inst(&a.t, "{op} {a.t} {a.n}, {b.n}")
	}

	fun iadd(&mut self, a &V, b &V) V {
		return self.bin("add", a, b)
	}

	fun isub(&mut self, a &V, b &V) V {
		return self.bin("sub", a, b)
	}

	fun imul(&mut self, a &V, b &V) V {
		return self.bin("mul", a, b)
	}

	fun band(&mut self, a &V, b &V) V {
		return self.bin("and", a, b)
	}

	fun bor(&mut self, a &V, b &V) V {
		return self.bin("or", a, b)
	}

	fun bxor(&mut self, a &V, b &V) V {
		return self.bin("xor", a, b)
	}

	fun iadd_imm(&mut self, a &V, k i64) V {
		if k == 0 {
			return a.clone()
		}
		return self.inst(&a.t, "add {a.t} {a.n}, {k}")
	}

	fun imul_imm(&mut self, a &V, k i64) V {
		return self.inst(&a.t, "mul {a.t} {a.n}, {k}")
	}

	fun bxor_imm(&mut self, a &V, k i64) V {
		return self.inst(&a.t, "xor {a.t} {a.n}, {k}")
	}

	fun ineg(&mut self, a &V) V {
		return self.inst(&a.t, "sub {a.t} 0, {a.n}")
	}

	fun bnot(&mut self, a &V) V {
		return self.inst(&a.t, "xor {a.t} {a.n}, -1")
	}

	// A shift amount of `a`'s type, masked to its width (as Cranelift does).
	fun shift_amount(&mut self, a &V, b &V) V {
		amt := self.resize(b, &a.t)
		return self.inst(&a.t, "and {a.t} {amt.n}, {bits(&a.t) - 1}")
	}

	fun shift(&mut self, op &str, a &V, b &V) V {
		amt := self.shift_amount(a, b)
		return self.inst(&a.t, "{op} {a.t} {a.n}, {amt.n}")
	}

	fun shift_imm(&mut self, op &str, a &V, k i64) V {
		return self.inst(&a.t, "{op} {a.t} {a.n}, {k}")
	}

	// Zero-extends or truncates an integer to type `t`.
	fun resize(&mut self, v &V, t &str) V {
		if bits(&v.t) == bits(t) {
			return v.clone()
		}
		if bits(&v.t) < bits(t) {
			return self.uextend(t, v)
		}
		return self.ireduce(t, v)
	}

	fun uextend(&mut self, t &str, v &V) V {
		if v.t == t {
			return v.clone()
		}
		return self.inst(t, "zext {v.t} {v.n} to {t}")
	}

	fun sextend(&mut self, t &str, v &V) V {
		if v.t == t {
			return v.clone()
		}
		return self.inst(t, "sext {v.t} {v.n} to {t}")
	}

	fun ireduce(&mut self, t &str, v &V) V {
		if v.t == t {
			return v.clone()
		}
		return self.inst(t, "trunc {v.t} {v.n} to {t}")
	}

	fun bool8(&mut self, i1 &str) V {
		return self.inst("i8", "zext i1 {i1} to i8")
	}

	// `cc`: eq, ne, slt, sle, sgt, sge, ult, ule, ugt, uge. The result is an i8 0/1.
	fun icmp(&mut self, cc &str, a &V, b &V) V {
		c := self.tmp()
		self.emit("{c} = icmp {cc} {a.t} {a.n}, {b.n}")
		return self.bool8(&c)
	}

	fun icmp_imm(&mut self, cc &str, a &V, k i64) V {
		c := self.tmp()
		self.emit("{c} = icmp {cc} {a.t} {a.n}, {k}")
		return self.bool8(&c)
	}

	// `cc`: oeq, une, olt, ole, ogt, oge.
	fun fcmp(&mut self, cc &str, a &V, b &V) V {
		c := self.tmp()
		self.emit("{c} = fcmp {cc} {a.t} {a.n}, {b.n}")
		return self.bool8(&c)
	}

	fun truthy(&mut self, c &V) string {
		n := self.tmp()
		self.emit("{n} = icmp ne {c.t} {c.n}, 0")
		return n
	}

	fun select(&mut self, c &V, a &V, b &V) V {
		k := self.truthy(c)
		return self.inst(&a.t, "select i1 {k}, {a.t} {a.n}, {a.t} {b.n}")
	}

	fun fbin(&mut self, op &str, a &V, b &V) V {
		return self.inst(&a.t, "{op} {a.t} {a.n}, {b.n}")
	}

	fun fneg(&mut self, a &V) V {
		return self.inst(&a.t, "fneg {a.t} {a.n}")
	}

	fun fpromote(&mut self, v &V) V {
		if v.t == "double" {
			return v.clone()
		}
		return self.inst("double", "fpext float {v.n} to double")
	}

	fun fdemote(&mut self, v &V) V {
		if v.t == "float" {
			return v.clone()
		}
		return self.inst("float", "fptrunc double {v.n} to float")
	}

	fun bitcast(&mut self, t &str, v &V) V {
		return self.inst(t, "bitcast {v.t} {v.n} to {t}")
	}

	fun llvm_intr(&mut self, name &str, ret &str, args &Vec<V>) V {
		ps mut := string::new()
		as_ mut := string::new()
		for (i, a) in args.iter().enumerate() {
			if i > 0 {
				ps.push(", ")
				as_.push(", ")
			}
			ps.push(&a.t)
			as_.push("{a.t} {a.n}")
		}
		self.declare(name, "declare {ret} @{name}({ps})")
		return self.inst(ret, "call {ret} @{name}({as_})")
	}

	fun intr_suffix(t &str) string {
		if t == "double" {
			return "f64".to_string()
		}
		if t == "float" {
			return "f32".to_string()
		}
		return t.to_string()
	}

	fun unary_intr(&mut self, base &str, v &V) V {
		name := "llvm.{base}.{Cg::intr_suffix(&v.t)}"
		args := one(v.clone())
		return self.llvm_intr(&name, &v.t, &args)
	}

	// High half of a full-width product.
	fun mulhi(&mut self, a &V, b &V, signed bool) V {
		w := int_of_bits(bits(&a.t) * 2)
		(x, y) := if signed { (self.sextend(&w, a), self.sextend(&w, b)) } else { (self.uextend(&w, a), self.uextend(&w, b)) }
		p := self.imul(&x, &y)
		h := self.shift_imm("lshr", &p, bits(&a.t) as i64)
		return self.ireduce(&a.t, &h)
	}

	fun isplit(&mut self, v &V) (V, V) {
		lo := self.ireduce("i64", v)
		h := self.shift_imm("lshr", v, 64)
		hi := self.ireduce("i64", &h)
		return (lo, hi)
	}

	// ---- control flow ----

	fun jump(&mut self, b u32) {
		self.term("br label %b{b}")
	}

	fun brif(&mut self, c &V, t u32, f u32) {
		k := self.truthy(c)
		self.term("br i1 {k}, label %b{t}, label %b{f}")
	}

	fun trap(&mut self) {
		self.declare("llvm.trap", "declare void @llvm.trap() cold noreturn nounwind")
		self.emit("call void @llvm.trap()")
		self.term("unreachable".to_string())
	}

	fun ret_void(&mut self) {
		self.term("ret void".to_string())
	}

	// `when(cond) { ... }`: returns the join block, entered after `when_end`.
	fun when_begin(&mut self, c &V) u32 {
		yes := self.new_block()
		done := self.new_block()
		self.brif(c, yes, done)
		self.switch_to(yes)
		return done
	}

	fun when_end(&mut self, done u32) {
		if self.open {
			self.jump(done)
		}
		self.switch_to(done)
	}

	// `fail_if(cond) { panic }`: the panic path ends in a trap.
	fun fail_begin(&mut self, c &V) u32 {
		yes := self.new_block()
		ok := self.new_block()
		self.brif(c, yes, ok)
		self.switch_to(yes)
		return ok
	}

	fun fail_end(&mut self, ok u32) {
		if self.open {
			self.trap()
		}
		self.switch_to(ok)
	}

	// A counted loop `for i in 0..n`: returns (i, head, done); the body follows.
	fun loop_begin(&mut self, n &V) (V, V, u32, u32) {
		i := self.slot(8, 8)
		z := self.iconst(0)
		self.store(&z, &i, 0)
		head := self.new_block()
		body := self.new_block()
		done := self.new_block()
		self.jump(head)
		self.switch_to(head)
		iv := self.load("i64", &i, 0)
		more := self.icmp("ult", &iv, n)
		self.brif(&more, body, done)
		self.switch_to(body)
		return (i, iv, head, done)
	}

	fun loop_end(&mut self, i &V, iv &V, head u32, done u32) {
		next := self.iadd_imm(iv, 1)
		self.store(&next, i, 0)
		self.jump(head)
		self.switch_to(done)
	}

	// ---- calls ----

	fun args_text(args &Vec<V>) string {
		o mut := string::new()
		for (i, a) in args.iter().enumerate() {
			if i > 0 {
				o.push(", ")
			}
			o.push("{a.t} {a.n}")
		}
		return o
	}

	fun call_rt(&mut self, name &str, args &Vec<V>) option<V> {
		self.declare_rt(name)
		(ps, r) := runtime_sig(name).unwrap()
		// Coerce arguments to the declared types.
		vals mut := Vec::new()
		for (i, a) in args.iter().enumerate() {
			p := &ps[i]
			if a.t == *p || is_float_t(p) {
				vals.push(a.clone())
			} else {
				vals.push(self.resize(a, p))
			}
		}
		at := Cg::args_text(&vals)
		if r == "void" {
			self.emit("call void @{name}({at})")
			return None
		}
		return Some(self.inst(&r, "call {r} @{name}({at})"))
	}

	fun call_fn(&mut self, f &str, args &Vec<V>) {
		at := Cg::args_text(args)
		self.emit("call void @{f}({at})")
	}

	fun call_indirect(&mut self, code &V, args &Vec<V>) {
		p := self.tmp()
		self.emit("{p} = inttoptr i64 {code.n} to ptr")
		at := Cg::args_text(args)
		self.emit("call void {p}({at})")
	}

	fun func_addr(&mut self, f &str) V {
		return self.inst("i64", "ptrtoint ptr @{f} to i64")
	}

	fun global_addr(&mut self, g &str) V {
		return self.inst("i64", "ptrtoint ptr @{g} to i64")
	}

	fun str_lit(&mut self, s &str) (V, V) {
		g := self.string_data(s)
		p := self.global_addr(&g)
		n := self.iconst(s.len() as i64)
		return (p, n)
	}

	fun panic_msg(&mut self, msg &str) {
		(p, n) := self.str_lit(msg)
		self.call_rt("kn_panic", &vec2(p, n))
	}
}

fun vec2(a V, b V) Vec<V> {
	o mut := Vec::new()
	o.push(a)
	o.push(b)
	return o
}

fun vec3(a V, b V, c V) Vec<V> {
	o mut := vec2(a, b)
	o.push(c)
	return o
}

fun vec4(a V, b V, c V, d V) Vec<V> {
	o mut := vec3(a, b, c)
	o.push(d)
	return o
}
