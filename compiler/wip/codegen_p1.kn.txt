// Native code generation: a port of crates/kiln_codegen that writes LLVM IR
// as text (compiled and linked by `clang`) instead of driving Cranelift.
//
// The translation mirrors the Rust backend's Cranelift lowering one to one:
// every MIR local lives in a stack slot, values move through memory, and all
// Kiln functions take `(ret, args...)` as pointers to caller-owned slots
// (closures take an environment pointer after `ret`). Addresses are `i64`
// values, as in Cranelift IR; loads and stores convert them to `ptr`.
// Release-only optimizations (register ABI, inlining, slot promotion, bounds
// check elimination) are left to LLVM.

// A value: its LLVM name (or literal) and type (`i8` .. `i128`, `float`, `double`).
@derive(Clone)
pub struct V {
	n string,
	t string,
}

fun val(n string, t &str) V {
	return V { n: n, t: t.to_string() }
}

fun bits(t &str) u32 {
	match t {
		"i8" => return 8,
		"i16" => return 16,
		"i32" => return 32,
		"i64" => return 64,
		"i128" => return 128,
		"float" => return 32,
		"double" => return 64,
		_ => return 64,
	}
}

fun is_float_t(t &str) bool {
	return t == "float" || t == "double"
}

fun int_of_bits(n u32) string {
	return "i{n}"
}

@derive(Clone, Copy, Eq)
pub enum Abi {
	Plain,
	Closure,
}

@derive(Clone)
pub struct Layout {
	pub size u32,
	pub align u32,
}

fun lay(size u32, align u32) Layout {
	return Layout { size: size, align: align }
}

fun align_up(x u32, a u32) u32 {
	return (x + a - 1) / a * a
}

// Offsets of fields laid out in order, plus the total layout.
fun sequential(fields &Vec<Layout>) (Vec<u32>, Layout) {
	off mut := 0u32
	align mut := 1u32
	offs mut := Vec::new()
	for f in fields {
		off = align_up(off, f.align)
		offs.push(off)
		off += f.size
		if f.align > align {
			align = f.align
		}
	}
	return (offs, lay(align_up(off, align), align))
}

fun hex_u64(x u64) string {
	digits := "0123456789ABCDEF"
	o mut := string::new()
	i mut := 15
	while true {
		d := ((x >> ((i * 4) as u64)) & 15) as usize
		o.push(&digits[d..d + 1])
		if i == 0 {
			break
		}
		i -= 1
	}
	return o
}

fun f64_bits(x f64) u64 {
	unsafe {
		p := &x as *const f64 as *const u64
		return std::ptr::read(p)
	}
}

fun f64_lit(x f64) string {
	return "0x{hex_u64(f64_bits(x))}"
}

// LLVM's byte-string constant body (`c"..."`).
fun ll_bytes(b &[u8]) string {
	digits := "0123456789ABCDEF"
	o mut := string::new()
	for c in b {
		x := *c
		if x >= 32 && x < 127 && x != 34 && x != 92 {
			o.push_char(x as char)
		} else {
			o.push("\\")
			hi := (x >> 4) as usize
			lo := (x & 15) as usize
			o.push(&digits[hi..hi + 1])
			o.push(&digits[lo..lo + 1])
		}
	}
	return o
}

fun clean_name(s &str) string {
	o mut := string::new()
	for c in s.chars() {
		if (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') {
			o.push_char(c)
		} else {
			o.push_char('_')
		}
	}
	return o
}

// Pending work: functions declared but not yet defined.
@derive(Clone)
enum Work {
	Fn { name string, fn_idx usize, body usize, substs HashMap<string, Ty>, abi Abi },
	Cmp(string, Ty),
	Debug(string, Ty),
	Hash(string, Ty),
	Eq(string, Ty),
	Clone(string, Ty),
	Drop(string, Ty),
	Shim { name string, target string, nargs usize },
}

pub struct LocalInfo {
	addr V,
	ty Ty,
	// Drop flag slot (absent for types without drop glue).
	flag option<V>,
	// Field paths moved out somewhere in the body, each with its flag.
	paths Vec<(Vec<string>, V)>,
}

// What a call resolves to after monomorphization.
enum Target {
	Direct(string),
	Virtual(usize),
	Intrinsic(string),
	Extern(string),
	Missing(string),
}

fun empty_body(span Span) Body {
	return Body {
		name: string::new(),
		kind: BodyKind::Fn,
		locals: Vec::new(),
		blocks: Vec::new(),
		arg_count: 0,
		return_flows: None,
		span: span,
		errors: Vec::new(),
		unsupported: Vec::new(),
		captures: Vec::new(),
		bounded_loops: Vec::new(),
	}
}

fun ty_key(t &Ty) string {
	return mir::ty(t)
}

fun subst_key(s &HashMap<string, Ty>) string {
	kv mut := Vec::new()
	for (k, v) in s {
		kv.push("{k}={ty_key(v)}")
	}
	kv.sort()
	o mut := string::new()
	for x in &kv {
		o.push(x)
		o.push(";")
	}
	return o
}

fun subst_map(s &Vec<(string, Ty)>) HashMap<string, Ty> {
	m mut := HashMap::new()
	for (k, v) in s {
		m.insert(k.clone(), v.clone())
	}
	return m
}

fun ptr_params(n usize) string {
	o mut := string::new()
	for i in 0..n {
		if i > 0 {
			o.push(", ")
		}
		o.push("i64 %p{i}")
	}
	return o
}

impl Cg {
	// ---- errors ----

	fun fatal(&mut self, msg string) {
		self.errors.push((msg, self.span))
	}

	fun error(&mut self, msg string) {
		self.errors.push((msg, self.span))
	}

	fun error_at(&mut self, msg string, span Span) {
		self.errors.push((msg, span))
	}

	fun fresh_name(&mut self, base &str) string {
		self.counter += 1
		return "kn{self.counter}_{clean_name(base)}"
	}

	fun fmt_ty(&self, t &Ty) string {
		return self.c.fmt_ty(t)
	}

	// ---- module-level declarations ----

	fun declare(&mut self, name &str, text string) {
		if !self.declared.contains(name) {
			self.declared.insert(name.to_string())
			self.decls.push(&text)
			self.decls.push("\n")
		}
	}

	fun declare_rt(&mut self, name &str) {
		match runtime_sig(name) {
			Some((ps, r)) => {
				ptxt mut := string::new()
				for (i, p) in ps.iter().enumerate() {
					if i > 0 {
						ptxt.push(", ")
					}
					ptxt.push(p)
				}
				attr := if name == "_setjmp" { " returns_twice" } else { "" }
				self.declare(name, "declare {r} @{name}({ptxt}){attr}")
			}
			None => {}
		}
	}

	fun string_data(&mut self, s &str) string {
		if let Some(n) = self.strings.get(s) {
			return n.clone()
		}
		name := self.fresh_name("str")
		self.globals.push("@{name} = private unnamed_addr constant [{s.len() + 1} x i8] c\"{ll_bytes(s.as_bytes())}\\00\", align 1\n")
		self.strings.insert(s.to_string(), name.clone())
		return name
	}

	fun bytes_data(&mut self, b &[u8]) string {
		name := self.fresh_name("bytes")
		self.globals.push("@{name} = private unnamed_addr constant [{b.len() + 1} x i8] c\"{ll_bytes(b)}\\00\", align 1\n")
		return name
	}

	fun static_data(&mut self, d u32) string {
		for (sd, n, _) in &self.statics {
			if *sd == d {
				return n.clone()
			}
		}
		fi mut := 0usize
		for (sd, i) in &self.c.r.statics {
			if *sd == d {
				fi = *i
			}
		}
		(_, t) := self.c.signature(fi)
		l := self.layout(&t)
		dn := self.c.r.def(d).name.clone()
		name := self.fresh_name("static_{dn}")
		size := if l.size == 0 { 1 } else { l.size }
		align := if l.align == 0 { 1 } else { l.align }
		self.globals.push("@{name} = internal global [{size} x i8] zeroinitializer, align {align}\n")
		self.statics.push((d, name.clone(), fi))
		return name
	}

	// ---- types and layout ----

	fun is_unsized(&self, t &Ty) bool {
		match t {
			Ty::Prim(p) => return p == "Str",
			Ty::Slice(_) | Ty::Dyn(_) => return true,
			_ => return false,
		}
	}

	fun is_lang(&self, t &Ty, name &str) bool {
		match t {
			Ty::Adt(d, _) => return self.c.r.prelude_type(name) == Some(*d),
			_ => return false,
		}
	}

	fun pointee(&self, t &Ty) option<Ty> {
		match t {
			Ty::Ref(_, x) => return Some((**x).clone()),
			Ty::Ptr(_, x) => return Some((**x).clone()),
			Ty::Adt(_, a) => {
				if self.is_lang(t, "Box") {
					return Some(a[0].clone())
				}
				return None
			}
			_ => return None,
		}
	}

	fun is_fat(&self, t &Ty) bool {
		match self.pointee(t) {
			Some(p) => return self.is_unsized(&p),
			None => return false,
		}
	}

	// The IR type of a scalar, or None for aggregates.
	fun scalar(&self, t &Ty) option<string> {
		match t {
			Ty::Prim(p) => {
				s := match p.as_str() {
					"I8" | "U8" | "Bool" => "i8",
					"I16" | "U16" => "i16",
					"I32" | "U32" | "Char" => "i32",
					"I64" | "U64" | "Isize" | "Usize" => "i64",
					"I128" | "U128" => "i128",
					"F32" => "float",
					"F64" => "double",
					_ => return None,
				}
				return Some(s.to_string())
			}
			Ty::Ref(_, _) | Ty::Ptr(_, _) => {
				if self.is_fat(t) {
					return None
				}
				return Some("i64".to_string())
			}
			Ty::Adt(_, _) => {
				if self.is_lang(t, "Box") && !self.is_fat(t) {
					return Some("i64".to_string())
				}
				return None
			}
			Ty::ChanSend(_) | Ty::ChanRecv(_) => return Some("i64".to_string()),
			_ => return None,
		}
	}

	fun layout(&mut self, t &Ty) Layout {
		key := ty_key(t)
		if let Some(l) = self.layouts.get(&key) {
			return l.clone()
		}
		l := self.compute_layout(t)
		self.layouts.insert(key, l.clone())
		return l
	}

	fun layouts_of(&mut self, ts &Vec<Ty>) Vec<Layout> {
		o mut := Vec::new()
		for t in ts {
			o.push(self.layout(t))
		}
		return o
	}

	fun field_layouts(&mut self, fs &Vec<(string, Ty)>) Vec<Layout> {
		o mut := Vec::new()
		for (_, t) in fs {
			o.push(self.layout(t))
		}
		return o
	}

	fun compute_layout(&mut self, t &Ty) Layout {
		if let Some(s) = self.scalar(t) {
			n := bits(&s) / 8
			return lay(n, if n < 8 { n } else { 8 })
		}
		match t {
			Ty::Unit | Ty::Never => return lay(0, 1),
			Ty::Ref(_, _) | Ty::Ptr(_, _) => return lay(16, 8),
			Ty::Fn(_, _, _) => return lay(16, 8),
			Ty::Tuple(ts) => {
				ls := self.layouts_of(ts)
				(_, l) := sequential(&ls)
				return l
			}
			Ty::ChanSend(_) | Ty::ChanRecv(_) => return lay(8, 8),
			Ty::Adt(d, a) => {
				if self.is_lang(t, "Box") {
					return lay(16, 8)
				}
				if self.is_lang(t, "Task") {
					return lay(8, 8)
				}
				if self.is_lang(t, "string") || self.is_lang(t, "Vec") {
					return lay(24, 8)
				}
				if self.is_lang(t, "Range") {
					e := self.layout(&a[0])
					ls mut := Vec::new()
					ls.push(e.clone())
					ls.push(e)
					ls.push(lay(1, 1))
					(_, l) := sequential(&ls)
					return l
				}
				if let DefKind::Enum = self.c.r.def(*d).kind {
					(_, l) := self.enum_layout(t)
					return l
				}
				fs := self.c.adt_fields(t, None)
				ls := self.field_layouts(&fs)
				(_, l) := sequential(&ls)
				return l
			}
			Ty::Prim(_) | Ty::Slice(_) | Ty::Dyn(_) => return lay(0, 1),
			Ty::Array(x, n) => {
				e := self.layout(x)
				return lay(e.size * (*n as u32), e.align)
			}
			_ => {
				name := self.fmt_ty(t)
				self.fatal("code generation does not support values of type `{name}` yet")
				return lay(8, 8)
			}
		}
	}

	// Payload offset and total layout of an enum.
	fun enum_layout(&mut self, t &Ty) (u32, Layout) {
		d := match t {
			Ty::Adt(d, _) => *d,
			_ => return (0, lay(0, 1)),
		}
		max_align mut := 4u32
		max_payload mut := 0u32
		for v in self.c.r.variants(d) {
			fs := self.c.adt_fields(t, Some(v))
			ls := self.field_layouts(&fs)
			(_, l) := sequential(&ls)
			if l.align > max_align {
				max_align = l.align
			}
			if l.size > max_payload {
				max_payload = l.size
			}
		}
		off := align_up(4, max_align)
		return (off, lay(align_up(off + max_payload, max_align), max_align))
	}

	fun variant_named(&self, d u32, name &str) option<u32> {
		for v in self.c.r.variants(d) {
			if self.c.r.def(v).name == name {
				return Some(v)
			}
		}
		return None
	}

	// Offset and type of field `name` of `t` (`"0"` for tuples, `"Variant.f"` for enum fields).
	fun field(&mut self, t &Ty, name &str) (u32, Ty) {
		match t {
			Ty::Tuple(ts) => {
				i := match parse_usize(name) {
					Some(i) => i,
					None => 0,
				}
				ls := self.layouts_of(ts)
				(offs, _) := sequential(&ls)
				if i < ts.len() {
					return (offs[i], ts[i].clone())
				}
				return (0, Ty::Unit)
			}
			Ty::Adt(d, a) => {
				if self.is_lang(t, "Range") {
					e := self.layout(&a[0])
					match name {
						"end" => return (align_up(e.size, e.align), a[0].clone()),
						"incl" => return (align_up(e.size, e.align) + e.size, prim("U8")),
						_ => return (0, a[0].clone()),
					}
				}
				match name.find(".") {
					Some(k) => {
						vname := &name[..k]
						f := &name[k + 1..]
						(poff, _) := self.enum_layout(t)
						v := self.variant_named(*d, vname)
						fields := self.c.adt_fields(t, v)
						ls := self.field_layouts(&fields)
						(offs, _) := sequential(&ls)
						for (i, (n, ft)) in fields.iter().enumerate() {
							if n == f {
								return (poff + offs[i], ft.clone())
							}
						}
						if fields.len() > 0 {
							return (poff + offs[0], fields[0].1.clone())
						}
						return (poff, Ty::Unit)
					}
					None => {
						fields := self.c.adt_fields(t, None)
						ls := self.field_layouts(&fields)
						(offs, _) := sequential(&ls)
						for (i, (n, ft)) in fields.iter().enumerate() {
							if n == name {
								return (offs[i], ft.clone())
							}
						}
						if fields.len() > 0 {
							return (offs[0], fields[0].1.clone())
						}
						return (0, Ty::Unit)
					}
				}
			}
			_ => return (0, Ty::Unit),
		}
	}

	fun variants_of(&self, t &Ty) Vec<option<u32>> {
		o mut := Vec::new()
		match t {
			Ty::Adt(d, _) => {
				if let DefKind::Enum = self.c.r.def(*d).kind {
					for v in self.c.r.variants(*d) {
						o.push(Some(v))
					}
					return o
				}
			}
			_ => {}
		}
		o.push(None)
		return o
	}

	// Whether dropping a `t` runs any code.
	fun needs_drop(&mut self, t &Ty) bool {
		key := ty_key(t)
		if let Some(b) = self.needs_drop_cache.get(&key) {
			return *b
		}
		self.needs_drop_cache.insert(key.clone(), false)
		b := match t {
			Ty::Adt(d, _) => {
				if self.is_lang(t, "string") || self.is_lang(t, "Vec") || self.is_lang(t, "Box") || self.is_lang(t, "Task") {
					true
				} else if self.is_lang(t, "Range") {
					false
				} else if self.c.has_drop_impl(t) {
					true
				} else {
					any mut := false
					for v in self.variants_of(t) {
						for (_, ft) in self.c.adt_fields(t, v) {
							if self.needs_drop(&ft) {
								any = true
							}
						}
					}
					any
				}
			}
			Ty::ChanSend(_) | Ty::ChanRecv(_) => true,
			Ty::Tuple(ts) => {
				any mut := false
				for x in ts {
					if self.needs_drop(x) {
						any = true
					}
				}
				any
			}
			Ty::Array(x, n) => *n > 0 && self.needs_drop(x),
			_ => false,
		}
		self.needs_drop_cache.insert(key, b)
		return b
	}

	// ---- instances ----

	fun bodies(&mut self, i usize) Vec<Body> {
		if !self.bodies.contains_key(&i) {
			b := lower::lower_fn(&mut self.c, i)
			self.bodies.insert(i, b)
		}
		return self.bodies.get(&i).unwrap().clone()
	}

	fun body_count_args(&mut self, i usize, bi usize) (usize, string) {
		if !self.bodies.contains_key(&i) {
			b := lower::lower_fn(&mut self.c, i)
			self.bodies.insert(i, b)
		}
		bs := self.bodies.get(&i).unwrap()
		if bi < bs.len() {
			return (bs[bi].arg_count, bs[bi].name.clone())
		}
		return (0, string::new())
	}

	// Declares (and queues for definition) an instance of body `body` of
	// function `fn_idx` under `substs`.
	fun instance(&mut self, fn_idx usize, body usize, substs HashMap<string, Ty>, abi Abi) string {
		a := if abi == Abi::Closure { "C" } else { "P" }
		key := "{fn_idx}/{body}/{a}/{subst_key(&substs)}"
		if let Some(n) = self.instances.get(&key) {
			return n.clone()
		}
		(_, base) := self.body_count_args(fn_idx, body)
		name := self.fresh_name(&base)
		self.instances.insert(key, name.clone())
		self.queue.push(Work::Fn { name: name.clone(), fn_idx: fn_idx, body: body, substs: substs, abi: abi })
		return name
	}

	// A closure-ABI shim around a plain function (fn items used as values).
	fun shim(&mut self, target &str, nargs usize) string {
		if let Some(n) = self.shims.get(target) {
			return n.clone()
		}
		name := self.fresh_name("shim")
		self.shims.insert(target.to_string(), name.clone())
		self.queue.push(Work::Shim { name: name.clone(), target: target.to_string(), nargs: nargs })
		return name
	}

	fun glue_name(&mut self, kind &str, t &Ty) string {
		key := ty_key(t)
		found := match kind {
			"drop" => self.drop_glue.get(&key).cloned(),
			"eq" => self.eq_glue.get(&key).cloned(),
			"cmp" => self.cmp_glue.get(&key).cloned(),
			"hash" => self.hash_glue.get(&key).cloned(),
			"debug" => self.debug_glue.get(&key).cloned(),
			_ => self.clone_glue.get(&key).cloned(),
		}
		if let Some(n) = found {
			return n
		}
		self.counter += 1
		name := "kn_{kind}{self.counter}"
		match kind {
			"drop" => {
				self.drop_glue.insert(key, name.clone())
				self.queue.push(Work::Drop(name.clone(), t.clone()))
			}
			"eq" => {
				self.eq_glue.insert(key, name.clone())
				self.queue.push(Work::Eq(name.clone(), t.clone()))
			}
			"cmp" => {
				self.cmp_glue.insert(key, name.clone())
				self.queue.push(Work::Cmp(name.clone(), t.clone()))
			}
			"hash" => {
				self.hash_glue.insert(key, name.clone())
				self.queue.push(Work::Hash(name.clone(), t.clone()))
			}
			"debug" => {
				self.debug_glue.insert(key, name.clone())
				self.queue.push(Work::Debug(name.clone(), t.clone()))
			}
			_ => {
				self.clone_glue.insert(key, name.clone())
				self.queue.push(Work::Clone(name.clone(), t.clone()))
			}
		}
		return name
	}

	fun drain(&mut self) {
		while self.qhead < self.queue.len() {
			if self.errors.len() > 20 {
				return
			}
			w := self.queue[self.qhead].clone()
			self.qhead += 1
			match w {
				Work::Fn { name, fn_idx, body, substs, abi } => self.define_fn(&name, fn_idx, body, substs, abi),
				Work::Drop(n, t) => self.define_glue(&n, "drop", &t),
				Work::Eq(n, t) => self.define_glue(&n, "eq", &t),
				Work::Cmp(n, t) => self.define_glue(&n, "cmp", &t),
				Work::Hash(n, t) => self.define_glue(&n, "hash", &t),
				Work::Debug(n, t) => self.define_glue(&n, "debug", &t),
				Work::Clone(n, t) => self.define_glue(&n, "clone", &t),
				Work::Shim { name, target, nargs } => self.define_shim(&name, &target, nargs),
			}
		}
	}
}

fun parse_usize(s &str) option<usize> {
	if s.len() == 0 {
		return None
	}
	n mut := 0usize
	for ch in s.chars() {
		if ch < '0' || ch > '9' {
			return None
		}
		n = n * 10 + (ch as usize - '0' as usize)
	}
	return Some(n)
}
